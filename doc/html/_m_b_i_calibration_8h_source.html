99 /*
100 * Newton Method is a numerical method looking for a root of a real - valued function
101 * Xn+1 = Xn - f(Xn) / f'(Xn)
102 * f(X) is the real-valued function and f'(X) is the derivate function
103 * Newton Method need a starting point X0 and then X1 is calculated from X0, X2 is calculated from X1, ...
104 * If the newton method is converging, Xn+1 is closer to the root than Xn, otherwise the method fails.
105 * Finally, if (Xn+1 - Xn) is smaller than a predefined value, the iteration stops and return Xn+1 as the root.
106 */
107 class MBI_DLLCPP NewtonMethod

109 public:
111 NewtonMethod(CalPolynomial* polynomial);
112
114 double RootStartFromX(double startingX);
115
117 bool isSuccess();

120 CalPolynomial* _polynomial;
